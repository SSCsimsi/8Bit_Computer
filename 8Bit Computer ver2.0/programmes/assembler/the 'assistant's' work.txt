#include <stdint.h>
#include <iostream>
#include <fstream>
#include <string>
#include <list>

using namespace std;

#define CMDLENGTH 3

class Cmd {
public:

	/*
	 * the name is made up of 3 ascii symbols,
	 * the value has to fit within 8 bits
	 */
	void Init(string nameIn, uint8_t valIn) {
		name = nameIn;
		val = valIn;
	}

	uint8_t Compare(string newLine) {
		return name == newLine.substr(0, CMDLENGTH);
	}

	uint8_t Put() {
		return val;
	}


private:
	string name;
	uint8_t val;
};

list<Cmd> Commands;

int main() {

	Cmd Nop;
	Nop.Init("nop", 0x00);
	Commands.push_back(Nop);

	Cmd Jump;
	Jump.Init("jmp", 0x01);
	Commands.push_back(Jump);

	Cmd JumpIf;
	JumpIf.Init("cjp", 0x02);
	Commands.push_back(JumpIf);

	/* Open the code */
	ifstream codeFile("code.txt");

	/* Create the hex files for programming the computer */
	ofstream cmdFile("cmd.hex"); //cmd and dat in one file, since they are combined in one flash chip
	ofstream adrFile("adr.hex");

	string newLine;
	uint8_t check;

	while(getline(codeFile, newLine)) {

		check = 0;

		for(Cmd cmd : Commands) {
			if(cmd.Compare(newLine)) {

				cmdFile << cmd.Put();
				check = 1;

				break;
			}
		}

		if(!check) {
			ofstream Error("UnknownCommand.txt");
			Error.close();

			return -1;
		}

		cmdFile << (uint8_t)stoi(newLine.substr(4, 5)); //value changed from string to int
		adrFile << (uint8_t)stoi(newLine.substr(7,10));

	}

	/* Close the files */
	codeFile.close();
	cmdFile.close();
	adrFile.close();
}

Oh this is a fun project 😄 — homemade 8-bit computer assemblers are chef’s kiss.
Yes, there are a few concrete problems here, and they explain exactly why your hex files don’t contain what you expect.

I’ll walk through the actual bugs, not just style stuff.

1. You are writing bytes as characters, not numbers

This is the main reason your .hex files look wrong.

cmdFile << cmd.Put();
cmdFile << (uint8_t)stoi(...);
adrFile << (uint8_t)stoi(...);


operator<< for ofstream treats uint8_t as a character, because uint8_t is typically just unsigned char.

So if cmd.Put() is 0x01, you are writing ASCII SOH, not the text "01" or the byte 0x01 in a binary sense.

What you probably want (pick one):
A) Write raw binary bytes

If these files are meant to be flashed directly:

uint8_t b = cmd.Put();
cmdFile.write(reinterpret_cast<char*>(&b), 1);


Same for the others:

uint8_t val = stoi(...);
cmdFile.write(reinterpret_cast<char*>(&val), 1);

uint8_t adr = stoi(...);
adrFile.write(reinterpret_cast<char*>(&adr), 1);


This writes actual 8-bit values, not ASCII.

B) Write textual hex (human-readable .hex)

If you want files like:

00
01
FF


Then do this:

cmdFile << hex << setw(2) << setfill('0') << (int)cmd.Put() << '\n';


and similarly for the others.
You’ll need:

#include <iomanip>

2. Compare() returns uint8_t instead of bool

This works accidentally, but it’s wrong and confusing:

uint8_t Compare(string newLine)


Should be:

bool Compare(const string& newLine)


Right now you’re returning either 0 or 1 as a byte, which is legal but brittle.

3. You silently assume every line is long enough
newLine.substr(0, CMDLENGTH)
newLine.substr(4, 5)
newLine.substr(7, 10)


If a line is shorter than expected → undefined behavior / exceptions.

At minimum:

if (newLine.length() < 10) {
    // error
}

4. stoi(substr(pos, len)) does NOT do what you think

This is subtle but important:

stoi(newLine.substr(4, 5));


The second argument is length, not end index.

So this grabs characters:

index 4, 5 characters long


Same problem here:

substr(7,10)


That’s 10 characters, not “up to index 10”.

If your line looks like:

jmp 12 34
012345678


You probably want:

stoi(newLine.substr(4, 2));
stoi(newLine.substr(7, 2));


(or better: parse with a stringstream)

5. You are skipping every second line (!)

This is a logic bug:

while (getline(codeFile, newLine)) {
    ...
    getline(codeFile, newLine);
}


That second getline eats the next line, so your assembler processes only every other line.

Delete it.

6. Your error handling overwrites the file every time

Not fatal, but odd:

ofstream Error("UnknownCommand.txt");


This truncates the file each time. If you want to log multiple errors:

ofstream Error("UnknownCommand.txt", ios::app);

The core fix (TL;DR)

The biggest, file-corrupting issue is this:

You are writing uint8_t with <<, which writes ASCII characters, not byte values.

Switch to either:

ofstream::write() for binary output or

formatted hex output with iomanip

Once you fix that + remove the extra getline, your assembler should suddenly start behaving like an actual assembler instead of a haunted typewriter 👻

Nice instruction set 👀 — this is starting to look like a real CPU.

Below is your table converted into the exact same style as your CreateList() function: one Cmd, Init(...), push_back(...).

⚠️ Important note first
Your assembler assumes 3-character mnemonics (CMDLENGTH 3).
Many instructions in your table are longer (“STORE OUTPUTS”, “BITWISE XOR”, etc.), so I’ve used reasonable 3-letter abbreviations.
You can rename them however you like later — the opcodes match your table.

void CreateList() {

    Cmd cmd;

    // --- CORE ---
    cmd.Init("nop", 0x00); Commands.push_back(cmd);
    cmd.Init("jmp", 0x01); Commands.push_back(cmd);
    cmd.Init("cjp", 0x02); Commands.push_back(cmd);

    // --- BUS TRANSFER ---
    cmd.Init("sti", 0x03); Commands.push_back(cmd); // STORE interface reg 1
    cmd.Init("stj", 0x04); Commands.push_back(cmd); // STORE interface reg 2
    cmd.Init("ldi", 0x05); Commands.push_back(cmd); // LOAD interface regs

    // --- STORAGE ---
    cmd.Init("sto", 0x06); Commands.push_back(cmd);
    cmd.Init("lod", 0x07); Commands.push_back(cmd);

    // --- ALU ---
    cmd.Init("add", 0x08); Commands.push_back(cmd);
    cmd.Init("adc", 0x09); Commands.push_back(cmd); // add with carry
    cmd.Init("sub", 0x0A); Commands.push_back(cmd);

    // --- INCREMENTOR ---
    cmd.Init("inc", 0x0B); Commands.push_back(cmd);
    cmd.Init("dec", 0x0C); Commands.push_back(cmd);

    // --- LOGIC ---
    cmd.Init("bnt", 0x0D); Commands.push_back(cmd); // binary NOT
    cmd.Init("bor", 0x0E); Commands.push_back(cmd);
    cmd.Init("ban", 0x0F); Commands.push_back(cmd);
    cmd.Init("bxr", 0x10); Commands.push_back(cmd);
    cmd.Init("not", 0x11); Commands.push_back(cmd); // bitwise
    cmd.Init("orr", 0x12); Commands.push_back(cmd);
    cmd.Init("and", 0x13); Commands.push_back(cmd);
    cmd.Init("xor", 0x14); Commands.push_back(cmd);

    // --- COMPARE ---
    cmd.Init("cgt", 0x15); Commands.push_back(cmd);
    cmd.Init("ceq", 0x16); Commands.push_back(cmd);
    cmd.Init("clt", 0x17); Commands.push_back(cmd);

    // --- A/B/C REGISTERS ---
    cmd.Init("sta", 0x18); Commands.push_back(cmd);
    cmd.Init("lda", 0x19); Commands.push_back(cmd);
    cmd.Init("stb", 0x1A); Commands.push_back(cmd);
    cmd.Init("ldb", 0x1B); Commands.push_back(cmd);
    cmd.Init("stc", 0x1C); Commands.push_back(cmd);
    cmd.Init("ldc", 0x1D); Commands.push_back(cmd);

    // --- FLAG REGISTER ---
    cmd.Init("stf", 0x1E); Commands.push_back(cmd);
    cmd.Init("ldc", 0x1F); Commands.push_back(cmd); // load carry
    cmd.Init("ldo", 0x20); Commands.push_back(cmd); // load overflow
    cmd.Init("ldf", 0x21); Commands.push_back(cmd);

    // --- SHIFT REGISTER ---
    cmd.Init("std", 0x22); Commands.push_back(cmd);
    cmd.Init("ldd", 0x23); Commands.push_back(cmd);
    cmd.Init("shf", 0x24); Commands.push_back(cmd);
    cmd.Init("shr", 0x25); Commands.push_back(cmd); // circular

    // --- INPUTS ---
    cmd.Init("sti", 0x26); Commands.push_back(cmd);
    cmd.Init("ldi", 0x27); Commands.push_back(cmd);
    cmd.Init("lur", 0x28); Commands.push_back(cmd); // load unread

    // --- OUTPUTS ---
    cmd.Init("sto", 0x29); Commands.push_back(cmd);
    cmd.Init("ldo", 0x2A); Commands.push_back(cmd);

    // --- SCREEN ---
    cmd.Init("str", 0x2B); Commands.push_back(cmd);
    cmd.Init("ldr", 0x2C); Commands.push_back(cmd);

    // --- CONTROLLER ---
    cmd.Init("lin", 0x2D); Commands.push_back(cmd);
}
