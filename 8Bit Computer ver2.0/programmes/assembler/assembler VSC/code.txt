label: Restart
label: Start
label: Add
label: Sub
label: Mul
label: MulRet
label: Div
label: DivRet
label: Error



//------ setup -------------------------------------------
0000; nop Restart
0001; sta 00
0002; stb 00
0003; stc 00 
0004; std 00
0005; sop 00
0006; sti 00
0007; stf 00
0008; stx 00
0009; sty 00



// ------ wait for input ---------------------------------
000A; nop Start
000B; ldu
000C; sta
000D; not 
000E; cjp Start // loop back for inputs



// ------ store input value ------------------------------
// store values to ram
000F; ldc
0010; stx
0011; ldz
0012; ldi
0013; sto

// increment input counter
0014; inc
0015; stc

// return if there are less than 3 inputs
0016; ldc
0017; sta
0018; stb 03
0019; clt
001A; cjp Start



// ------ decide calculation method -------------------------------

001B; lod 0001 // the calculation mode
001C; sta

// if (mode == 0)
001D; stb 00
001E; ceq
001F; cjp Add

// if (mode == 1)
0020; stb 01
0021; ceq
0022; cjp Sub

// if (mode == 2)
0023; stb 02
0024; ceq
0025; cjp Mul

// if (mode == 3)
0026; stb 03
0027; ceq
0028; cjp Div

// else (error detected)
0029; jmp Error



// ------ add -----------------------------------------------------
002A; nop Add
002B; lod 0000
002C; sta
002D; lod 0002
002E; stb
002F; add

0030; sop
0031; jmp Start



// ------ sub -----------------------------------------------------
0032; nop Sub
0033; lod 0000
0034; sta
0035; lod 0002
0036; stb
0037; sub

0038; sop
0039; jmp Start




// ------ mul ------------------------------------------------------

// ------------------------ memory map: ------------------------- //
//                                                                //
// 0000 > val 1                                                   //
// 0001 > operation                                               //
// 0002 > val 2                                                   //
//                                                                //
// 0010 > position of current bit being pulled from the number    //
// 0011 > temporary shift distance for val 1                      //
// 0012 > accumulator for the result                              //
//                                                                //
//                                                                //
//                                                                //
// -------------------------------------------------------------- //

// sft shifts in the wrong direction!!!

003A; nop Mul
003B; sto 01 0010
003C; nop
003D; sto 00 0012

// calc dist
003E; nop MulRet
003F; lod 0002
0040; sta
0041; lod 0010
0042; stb
0043; ban
0044; sta
0045; stb 01
0046; sub
// fix the shift direction
0047; sta
0048; stb 08
0049; sub
004A; sto 0011

// select next bit
004B; lod 0010
004C; sta
004D; std 07 // should be 01, but the shift register shifts in the wrong direction
004E; sft
004F; sto 0010

// add shifted value to accumulator
0050; lod 0000
0051; sta
0052; lod 0011
0053; std
0054; sft
0055; sta
0056; lod 0012
0057; stb
0058; add
0059; sto 0012

// keep looping until the last bit is checked
005A; lod 0010
005B; sta
005C; stb 80
005D; ceq
005E; sta
005F; not
0060; cjp MulRet

0061; lod 0012
0062; sop
0063; jmp Start



// ------ div -----------------------------------------------------
0064; nop Div
0065; jmp Start
