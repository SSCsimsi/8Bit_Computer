//------ setup -------------------------------------------
0000; nop 
0001; sta 00
0002; stb 00
0003; stc 00 
0004; std 00
0005; sop 00
0006; sti 00
0007; stf 00
0008; stx 00
0009; sty 00
000A; nop start



// ------ wait for input ---------------------------------
000B; nop input
000C; ldu
000D; sta
000E; not 
000F; cjp input // /\ loop back for inputs /\



// ------ store input value ------------------------------
// increment input counter
0010; nop
0011; inc 
0012; stc



// store values to ram
0013; ldc
0014; stx
0015; ldz
0016; ldi
0017; sto


// ------ decide calculation method -------------------------------
0018; lod 0000 
0019; sta


// if (mode != 0)
001A; stb 00
001B; ceq
001C; sta
001D; not
001E; cjp skipadd 
// skip if (mode != 0)

// add
001F; nop  
0020; nop
0021; nop
0022; jmp start


// if (mode != 1)
0023; nop skipadd
0024; stb 01
0025; ceq
0026; sta
0027; not
0028; cjp skipsub 
// skip if (mode != 1)

// subtract
0029; nop 
002A; nop
002B; nop
002C; jmp start


// if (mode != 2)
002D; nop skipsub
002E; stb 02
002F; ceq
0030; sta
0031; not
0032; cjp skipmult 
//skip if (mode != 2)

// multiply
0033; nop 
0034; nop
0035; nop
0036; jmp start


// else divide
0037; nop skipmult

0039; nop
003A; jmp start
