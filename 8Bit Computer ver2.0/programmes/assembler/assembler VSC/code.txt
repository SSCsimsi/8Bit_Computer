//------ setup -------------------------------------------

0000; nop Restart
0001; sta 00
0002; stb 00
0003; stc 00 
0004; std 00
0005; sop 00
0006; sti 00
0007; stf 00
0008; stx 00
0009; sty 00

// input values
000A; sto 00 0000
000B; sto 00 0001
000C; sto 00 0002

// values for multiplication
000D; sto 00 0010
000E; sto 00 0011
000F; sto 00 0012



// ------ input handling ------------------------------------------

// wait for button to not be pressed anymore
0010; nop WaitForRelease
0011; ldu
0012; cjp WaitForRelease

// wait for input
0013; nop WaitForInput
0014; ldu
0015; sta
0016; not 
0017; cjp WaitForInput

// store values to ram
0018; ldc
0019; stx
001A; ldz
001B; ldi
001C; sto

// increment input counter
001D; inc
001E; stc

// return if there are less than 3 inputs
001F; ldc
0020; sta
0021; stb 03
0022; clt
0023; cjp WaitForRelease



// ------ decide calculation method -------------------------------

0024; lod 0001 // the calculation mode
0025; sta

// if (mode == 0)
0026; stb 00
0027; ceq
0028; cjp Add

// if (mode == 1)
0029; stb 01
002A; ceq
002B; cjp Sub

// if (mode == 2)
002C; stb 02
002D; ceq
002E; cjp Mul

// if (mode == 3)
002F; stb 03
0030; ceq
0031; cjp Div

// else (error detected)
0032; jmp Error



// ------ add -----------------------------------------------------

0033; nop Add
0034; lod 0000
0035; sta
0036; lod 0002
0037; stb
0038; add

0039; sop
003A; jmp Restart



// ------ sub -----------------------------------------------------

003B; nop Sub
003C; lod 0000
003D; sta
003E; lod 0002
003F; stb
0040; sub

0041; sop
0042; jmp Restart



// ------ mul ------------------------------------------------------

// ------------------------ memory map: ------------------------- //
//                                                                //
// 0000 > val 1                                                   //
// 0001 > operation                                               //
// 0002 > val 2                                                   //
//                                                                //
// 0010 > position of current bit being pulled from the number    //
// 0011 > temporary shift distance for val 1                      //
// 0012 > accumulator for the result                              //
//                                                                //
//                                                                //
//                                                                //
// -------------------------------------------------------------- //

// sft shifts in the wrong direction!!!

0043; nop Mul
0044; sto 01 0010
0045; nop
0046; sto 00 0012

// calc dist
0047; nop MulRet
0048; lod 0002
0049; sta
004A; lod 0010
004B; stb
004C; ban
004D; sta
004E; stb 01
004F; sub
// fix the shift direction
0050; stb
0051; sta 08
0052; sub
0053; sto 0011

// select next bit
0054; lod 0010
0055; sta
0056; std 07 // should be 01, but the shift register shifts in the wrong direction
0057; sft
0058; sto 0010

// add shifted value to accumulator
0059; lod 0000
005A; sta
005B; lod 0011
005C; std
005D; sft
005E; sta
005F; lod 0012
0060; stb
0061; add
0062; sto 0012

// keep looping until the last bit is checked
0063; lod 0010
0064; sta
0065; stb 80
0066; ceq
0067; sta
0068; not
0069; cjp MulRet

006A; lod 0012
006B; sop
006C; jmp Restart



// ------ div -----------------------------------------------------

006D; nop Div
006E; jmp Restart
