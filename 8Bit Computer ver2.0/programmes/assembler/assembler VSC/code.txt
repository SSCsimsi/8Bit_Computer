label: input
label: skipadd
label: skipsub
label: skipmult



//------ setup -------------------------------------------
0000; nop start
0001; sta 00
0002; stb 00
0003; stc 00 
0004; std 00
0005; sop 00
0006; sti 00
0007; stf 00
0008; stx 00
0009; sty 00



// ------ wait for input ---------------------------------
000A; nop input
000B; ldu
000C; sta
000D; not 
000E; cjp input // /\ loop back for inputs /\



// ------ store input value ------------------------------
// increment input counter
000F; nop
0010; inc 
0011; stc



// store values to ram
0012; ldc
0013; stx
0014; ldz
0015; ldi



// ------ decide calculation method -------------------------------
0016; lod 0000 
0017; sta


// if (mode != 0)
0018; stb 00
0019; ceq
001A; sta
001B; not
001C; cjp skipadd 
// skip if (mode != 0)

// add
001D; nop  
001E; nop
001F; nop
0020; jmp start


// if (mode != 1)
0021; nop skipadd
0022; stb 01
0023; ceq
0024; sta
0025; not
0026; cjp skipsub 
// skip if (mode != 1)

// subtract
0027; nop 
0028; nop
0029; nop
002A; jmp start


// if (mode != 2)
002B; nop skipsub
002C; stb 02
002D; ceq
002E; sta
002F; not
0030; cjp skipmult 
//skip if (mode != 2)

// multiply
0031; nop 
0032; nop
0033; nop
0034; jmp start


// else divide
0035; nop skipmult
0036; nop
0037; nop
0038; jmp start
