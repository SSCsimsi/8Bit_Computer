label: Restart
label: Start
label: Add
label: Sub
label: Mul
label: MulRet
label: Div
label: DivRet
label: Error



//------ setup -------------------------------------------
0000; nop Restart
0001; sta 00
0002; stb 00
0003; stc 00 
0004; std 00
0005; sop 00
0006; sti 00
0007; stf 00
0008; stx 00
0009; sty 00



// ------ wait for input ---------------------------------
000A; nop Start
000B; ldu
000C; sta
000D; not 
000E; cjp Start // loop back for inputs



// ------ store input value ------------------------------
// store values to ram
000F; ldc
0010; stx
0011; ldz
0012; ldi
0013; sto

// increment input counter
0014; inc
0015; stc

// return if there are less than 3 inputs
0016; ldc
0017; sta
0018; stb 03
0019; clt
001A; cjp Start



// ------ decide calculation method -------------------------------

001B; lod 0001 // the calculation mode
001C; sta

// if (mode == 0)
001D; stb 00
001E; ceq
001F; cjp Add

// if (mode == 1)
0020; stb 01
0021; ceq
0022; cjp Sub

// if (mode == 2)
0023; stb 02
0024; ceq
0025; cjp Mul

// if (mode == 3)
0026; stb 03
0027; ceq
0028; cjp Div

// else (error detected)
0029; jmp Error



// ------ add -----------------------------------------------------
002A; nop Add
002B; lod 0000
002C; sta
002D; lod 0002
002E; stb
002F; add

0030; sop
0031; jmp Start



// ------ sub -----------------------------------------------------
0032; nop Sub
0033; lod 0000
0034; sta
0035; lod 0002
0036; stb
0037; sub

0038; sop
0039; jmp Start




// ------ mul ------------------------------------------------------

// ------------------------ memory map: ------------------------- //
//                                                                //
// 0000 > val 1                                                   //
// 0001 > operation                                               //
// 0002 > val 2                                                   //
//                                                                //
// 0010 > position of current bit being pulled from the number    //
// 0011 > temporary shift distance for val 1                      //
// 0012 > accumulator for the result                              //
//                                                                //
//                                                                //
//                                                                //
// -------------------------------------------------------------- //

// sft shifts in the wrong direction!!!

003A; nop Mul
003B; sto 01 0010
003C; sto 00 0012

// calc dist
003D; nop MReturn
003E; lod 0002
003F; sta
0040; lod 0010
0041; stb
0042; ban
0043; sta
0044; stb 01
0045; sub
0046; sto 0011

// select next bit
0047; lod 0010
0048; sta
0049; std 01
004A; sft
004B; sto 0010

// add shifted value to accumulator
004C; lod 0000
004D; sta
004E; lod 0011
004F; std
0050; sft
0051; sta
0052; lod 0012
0053; stb
0054; add
0055; sto 0012

// keep looping until the last bit is checked
0056; lod 0010
0057; sta
0058; stb 80
0059; ceq
005A; sta
005B; not
005C; cjp MulRet

005D; lod 0012
005E; sop
005F; jmp Start



// ------ div -----------------------------------------------------
0060; nop Div
0061; jmp Start
