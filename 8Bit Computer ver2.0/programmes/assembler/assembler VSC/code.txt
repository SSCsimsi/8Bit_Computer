label: Start
label: Add
label: Sub

label: Div



//------ setup -------------------------------------------
0000; nop 
0001; sta 00
0002; stb 00
0003; stc 00 
0004; std 00
0005; sop 00
0006; sti 00
0007; stf 00
0008; stx 00
0009; sty 00



// ------ wait for input ---------------------------------
000A; nop Start
000B; ldu
000C; sta
000D; not 
000E; cjp Start // loop back for inputs



// ------ store input value ------------------------------
// increment input counter
000F; nop
0010; inc
0011; stc

// store values to ram
0012; ldc
0013; stx
0014; ldz
0015; ldi
0016; sto

// return if there are less than 3 inputs
0017; ldc
0018; sta
0019; stb 03
001A; clt
001B; cjp Start



// ------ decide calculation method -------------------------------

001C; lod 0001 // the calculation mode
001D; sta

// if (mode == 0)
001E; stb 00
001F; ceq
0020; cjp Add

// if (mode == 1)
0021; stb 01
0022; ceq
0023; cjp Sub

// if (mode == 2)
0024; stb 02
0025; ceq
0026; cjp Mul

// if (mode == 3)
0027; stb 03
0028; ceq
0029; cjp Div




// ------ add -----------------------------------------------------
002A; nop Add
002B; lod 0000
002C; sta
002D; lod 0002
002E; stb
002F; add
0030; sop
0031; jmp Start


// ------ sub -----------------------------------------------------
0032; nop Sub
0033; lod 0000
0034; sta
0035; lod 0002
0036; stb
0037; sub
0038; sop
0039; jmp Start



// ------ mul -----------------------------------------------------
003A; nop Mul
003B; jmp Start



// ------ div -----------------------------------------------------
003C; nop Div
003D; jmp Start
