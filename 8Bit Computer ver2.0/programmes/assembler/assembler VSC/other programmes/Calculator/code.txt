//------ setup -------------------------------------------

// ------------------------ memory map: ------------------------- //
//                                                                //
// 0000 > val 1                                                   //
// 0001 > operation                                               //
// 0002 > val 2                                                   //
//                                                                //
// 0010 > position of current bit being pulled from the number    //
// 0011 > temporary shift distance for val 1                      //
// 0012 > accumulator for the result                              //
//                                                                //
//                                                                //
//                                                                //
// -------------------------------------------------------------- //

0000; nop Restart
0001; sta 00
0002; stb 00
0003; stc 00 
0004; std 00
0005; sop 00
0006; sti 00
0007; stf 00
0008; stx 00
0009; sty 00
000A; ldu
; nop

// input values
000B; sto 00 0000
000C; sto 00 0001
000D; sto 00 0002

// values for multiplication
000E; sto 00 0010
000F; sto 00 0011
0010; sto 00 0012



// ------ input handling ------------------------------------------

// wait for button to not be pressed anymore
0011; nop WaitForRelease
0012; ldu
0013; cjp WaitForRelease

// wait for input
0014; nop WaitForInput
0015; ldu
0016; sta
0017; not 
0018; cjp WaitForInput

0019; ldc
001A; sta

001B; stb 00
001C; ceq
001D; cjp StoreInput1
001E; stb 01
001F; ceq
0020; cjp StoreMode
0021; stb 02
0022; ceq
0023; cjp StoreInput2

0024; nop StoreInput1
0025; ldi
0026; sto 0000
0027; jmp StoringDone

0028; nop StoreMode
0029; ldi
002A; sto 0001
002B; jmp StoringDone

002C; nop StoreInput2
002D; ldi
002E; sto 0002
002F; jmp StoringDone

0030; nop StoringDone

// increment input counter
0031; inc
0032; stc

0033; ldc
0034; sta
0035; stb 03
0036; clt
0037; cjp WaitForRelease

// ------ decide calculation method -------------------------------

0038; lod 0001 // the calculation mode
0039; sta

// if (mode == 0)
003A; stb 00
003B; ceq
003C; cjp Add

// if (mode == 1)
003D; stb 01
003E; ceq
003F; cjp Sub

// if (mode == 2)
0040; stb 02
0041; ceq
0042; cjp Mul

// if (mode == 3)
0043; stb 03
0044; ceq
0045; cjp Div

// else (error detected)
0046; jmp Error



// ------ add -----------------------------------------------------

0047; nop Add
0048; lod 0000
0049; sta
004A; lod 0002
004B; stb
004C; add

004D; sop
004E; jmp Wait



// ------ sub -----------------------------------------------------

004F; nop Sub
0050; lod 0000
0051; sta
0052; lod 0002
0053; stb
0054; sub

0055; sop
0056; jmp Wait



// ------ mul ------------------------------------------------------

// ------------------------ memory map: ------------------------- //
//                                                                //
// 0000 > val 1                                                   //
// 0001 > operation                                               //
// 0002 > val 2                                                   //
//                                                                //
// 0010 > position of current bit being pulled from the number    //
// 0011 > temporary shift distance for val 1                      //
// 0012 > accumulator for the result                              //
//                                                                //
//                                                                //
//                                                                //
// -------------------------------------------------------------- //

// sft shifts in the wrong direction!!!

0057; nop Mul
0058; sto 01 0010
0059; nop
005A; sto 00 0012

// calc dist
005B; nop MulRet
005C; lod 0002
005D; sta
005E; lod 0010
005F; stb
0060; ban
0061; sta
0062; stb 00
0063; ceq
0064; cjp MulAddSkip
0065; lda
0066; stc
0067; dec
// fix shift direction
0068; stb
0069; sta 07
006A; sub
006B; sto 0011

// add shifted value to accumulator
006C; lod 0000
006D; sta
006E; lod 0011
006F; std
0070; sfc
0071; sta
0072; lod 0012
0073; stb
0074; add
0075; sto 0012

// select next bit
0076; nop MulAddSkip
0077; lod 0010
0078; sta
0079; std 07 // shift by one
007A; sfc
007B; sto 0010

// keep looping until the last bit is checked
007C; lod 0010
007D; sta
007E; stb 80
007F; ceq
0080; sta
0081; not
0082; cjp MulRet

0083; lod 0012
0084; sop
0085; jmp Wait



// ------ div -----------------------------------------------------

0086; nop Div
0087; jmp Wait



// ------ loop ----------------------------------------------------

0088; nop Wait
0089; ldu // incase it stayed high

// loop for a bit
008A; nop
008B; nop
008C; nop
008D; nop

// actually check the input
008E; ldu
008F; cjp Restart
0090; jmp Wait
